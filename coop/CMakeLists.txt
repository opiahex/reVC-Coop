if(NOT WIN32 OR NOT LIBRW_PLATFORM STREQUAL "D3D9")
    message(FATAL_ERROR "Initial VC:COOP native adapter requires Windows + librw D3D9")
endif()
file(GLOB ENET_SOURCES CONFIGURE_DEPENDS "vendor/enet/*.c")
add_library(coop-enet STATIC ${ENET_SOURCES})
target_include_directories(coop-enet PUBLIC vendor/enet/include)
target_link_libraries(coop-enet PUBLIC ws2_32 winmm)
add_library(coop-network STATIC net/Network.cpp net/BackgroundClient.cpp)
target_include_directories(coop-network PUBLIC .)
target_compile_features(coop-network PUBLIC cxx_std_20)
target_compile_definitions(coop-network PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(coop-network PUBLIC coop-enet Threads::Threads)
add_library(coop-ui STATIC ui/Renderer.cpp ui/Overlay.cpp ui/ChatSettings.cpp)
target_link_libraries(coop-ui PUBLIC coop-network d3d9 gdi32 user32)
target_compile_options(coop-ui PRIVATE /utf-8)
add_library(coop-engine STATIC Engine.cpp NativeWorld.cpp Diagnostics.cpp)
target_link_libraries(coop-engine PRIVATE coop-ui librw::librw dbghelp)
target_compile_definitions(coop-engine PRIVATE LIBRW USE_D3D9 CMAKE_BUILD AUDIO_MSS CMAKE_NO_AUTOLINK USE_OUR_VERSIONING)
target_compile_options(coop-engine PRIVATE /Zc:sizedDealloc- /utf-8)
target_include_directories(coop-engine PRIVATE "${PROJECT_SOURCE_DIR}")
foreach(DIR IN LISTS REVC_INCLUDES)
    target_include_directories(coop-engine PRIVATE "${PROJECT_SOURCE_DIR}/src/${DIR}")
endforeach()
target_link_libraries(reVC PRIVATE coop-engine)
add_executable(vc-coop-server server/Main.cpp)
target_link_libraries(vc-coop-server PRIVATE coop-network)
add_executable(coop-tests tests/Tests.cpp)
target_link_libraries(coop-tests PRIVATE coop-ui)
add_test(NAME coop-portable-network-d3d9 COMMAND coop-tests)

target_compile_options(coop-tests PRIVATE /utf-8)
